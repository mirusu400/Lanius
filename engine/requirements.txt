# Needs Python 3.13 or newer: on 3.12 mitmproxy pins typing-extensions<=4.14,
# which conflicts with the pydantic that mcp requires.
#
# Lower bounds are the versions this project is developed and tested against,
# not the oldest that might work. Loose bounds made pip walk the whole release
# history of mitmproxy and its transitive dependencies, which on Python 3.12
# ends in 'resolution-too-deep' after several minutes. There is no upper bound,
# so security updates still come through.

fastapi>=0.141
uvicorn[standard]>=0.52
mcp>=2.2
